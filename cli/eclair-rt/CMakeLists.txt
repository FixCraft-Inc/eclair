add_library(eclair_render_scene STATIC scene.cpp)
find_package(Threads REQUIRED)
target_include_directories(eclair_render_scene PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(eclair_render_scene PUBLIC eclair_ops PRIVATE Threads::Threads)
eclair_target_warnings(eclair_render_scene)

if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    add_library(eclair_render_ipc STATIC ipc.cpp)
    target_include_directories(eclair_render_ipc PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
    target_link_libraries(eclair_render_ipc PUBLIC eclair_ops)
    eclair_target_warnings(eclair_render_ipc)
    add_executable(eclair-rt main.cpp control.cpp)
    target_link_libraries(eclair-rt PRIVATE eclair_render_ipc eclair_render_scene)
    eclair_target_warnings(eclair-rt)
    set_target_properties(eclair-rt PROPERTIES RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR})
    if(ECLAIR_BUILD_TESTS)
        find_package(Python3 COMPONENTS Interpreter REQUIRED)
        add_test(NAME eclair.render_demo COMMAND ${Python3_EXECUTABLE}
            ${PROJECT_SOURCE_DIR}/tests/test_render_demo.py --binary $<TARGET_FILE:eclair-rt>)
        set_tests_properties(eclair.render_demo PROPERTIES TIMEOUT 90)
    endif()
    if(ECLAIR_BUILD_RENDER_DEMO)
        find_package(SDL2 REQUIRED)
        target_sources(eclair-rt PRIVATE presenter_sdl.cpp)
        target_link_libraries(eclair-rt PRIVATE SDL2::SDL2)
    else()
        target_sources(eclair-rt PRIVATE presenter_stub.cpp)
    endif()
    if(ECLAIR_ENABLE_CUDA)
        add_executable(eclair-rt-cuda-worker cuda_worker.cpp)
        target_link_libraries(eclair-rt-cuda-worker PRIVATE eclair_render_ipc eclair_render_scene eclair_cuda)
        eclair_target_warnings(eclair-rt-cuda-worker)
        set_target_properties(eclair-rt-cuda-worker PROPERTIES RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR})
    endif()
endif()
